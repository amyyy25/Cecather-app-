    function pueblarSelectCiudades() {
        const select = document.getElementById('select-ciudad');
        select.innerHTML = '<option value="">-- SELECCIONAR CIUDAD --</option>';

        for (const prov in BASE_CIUDADES) {
            const optgroup = document.createElement('optgroup');
            optgroup.label = prov;
            BASE_CIUDADES[prov].forEach(c => {
                const opt = document.createElement('option');
                opt.value = `${prov}|${c.nombre}`;
                opt.innerText = `${c.nombre} (${c.grados.toString().replace(/\B(?=(\d{3})+(?!\d))/g, ".")} GDB / ${c.temp})`;
                optgroup.appendChild(opt);
            });
            select.appendChild(optgroup);
        }

        const optManual = document.createElement('option');
        optManual.value = "MANUAL";
        optManual.innerText = "⚙️ OTRA CIUDAD (MANUAL)";
        select.appendChild(optManual);

    function actualizarClimaCiudad() {
        const val = document.getElementById('select-ciudad').value;
        const grupoManual = document.getElementById('grupo-input-manual-ciudad');

        if (val === "MANUAL") {
            grupoManual.style.display = "flex";
        } else {
            grupoManual.style.display = "none";
            if (!val) return;
            const [prov, ciudadNombre] = val.split('|');
            const ciudadData = BASE_CIUDADES[prov].find(c => c.nombre === ciudadNombre);

            if (ciudadData) {
                document.getElementById('clima-grados').value = ciudadData.grados.toString().replace(/\B(?=(\d{3})+(?!\d))/g, ".");
                document.getElementById('clima-temp').value = ciudadData.temp;
            }
        }
    }

    function exportarProyectoArchivo() {

    function importarProyectoArchivo(event) {
        const file = event.target.files[0];
        if (!file) return;

        const reader = new FileReader();
        reader.onload = function(e) {
            try {
                const contenido = e.target.result;
                const data = JSON.parse(contenido);
                cargarDatosEnPantalla(data);
                guardarEstadoLocal();
                generarTodo();
                alert("¡Proyecto importado correctamente!");
            } catch (err) {
                alert("El archivo seleccionado no es válido o está dañado.");
            }
            event.target.value = ''; 
        };
        reader.readAsText(file);
    }

    function formatearNumero(num) {
        if (isNaN(num) || num === null) return "0";
        return Math.round(num).toString().replace(/\B(?=(\d{3})+(?!\d))/g, ".");
    }

    function formatearDecimal(num) {
        if (isNaN(num) || num === null) return "0,00";
        return num.toFixed(2).replace(".", ",").replace(/\B(?=(\d{3})+(?!\d))/g, ".");
    }
    
    function calcularTotalesProyectoSilencioso(data) {
        let totalM2 = 0, totalWCol = 0, ml75 = 0, ml116 = 0, ml150 = 0, totalML = 0, udFolio = 0;
        const cronos = data.estancias ? data.estancias.length : 0;

        if (data.estancias) {
            data.estancias.forEach(est => {
                totalM2 += parseFloat(est.m2) || 0;
                if (est.folios) {
                    est.folios.forEach(fol => {
                        const ml = parseFloat(fol.ml) || 0;
                        const tipo = parseInt(fol.tipo) || 75;
                        totalML += ml;
                        udFolio++;
                        totalWCol += (ml * tipo);
                        if(tipo === 75) ml75 += ml;
                        if(tipo === 116) ml116 += ml;
                        if(tipo === 150) ml150 += ml;
                    });
                }
            });
        }
        const densMedia = totalM2 > 0 ? Math.round(totalWCol / totalM2) : 0;
        return { totalM2, densMedia, totalWCol, ml75, ml116, ml150, totalML, udFolio, cronos, fecha: data.fecha || '-' };
    }

    function cambiarColorTipo(selectElement) {
        const val = selectElement.value;
        selectElement.className = `fol-tipo txt-${val}`;
    }

    function actualizarFolio(input, estanciaId) {
        const ml = parseFloat(input.value) || 0;
        const row = input.closest('.folio-row');
        const modInput = row.querySelector('.fol-mod');
        modInput.value = Math.round(ml / 0.25);
        calcularEstancia(estanciaId);
    }

    function calcularEstancia(estanciaId) {
        const est = document.getElementById(`estancia-${estanciaId}`);
        if(!est) return;

        const m2 = parseFloat(est.querySelector('.est-m2').value) || 0;
        const folios = est.querySelectorAll('.folio-row');

        let totalWatts = 0;
        folios.forEach(fol => {
            const ml = parseFloat(fol.querySelector('.fol-ml').value) || 0;
            const tipo = parseInt(fol.querySelector('.fol-tipo').value) || 75;
            totalWatts += ml * tipo;
        });

        const liveWm2 = m2 > 0 ? Math.round(totalWatts / m2) : 0;
        est.querySelector(`#live-wm2-${estanciaId}`).innerText = formatearNumero(liveWm2);
        est.querySelector(`#live-wcol-${estanciaId}`).innerText = formatearNumero(totalWatts);
    }
